  const ahora = Date.now();

  if (hoy) return hoy.id;
  const proxima = sesiones.find((sesion) => sesion.iniciaEn && new Date(sesion.iniciaEn).getTime() > ahora);
  return (proxima ?? sesiones[sesiones.length - 1])?.id ?? '';
};
